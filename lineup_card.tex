% TV Thing's channel lineup, printed like a cable company's lineup card from
% the late seventies: cream card stock, brown ink, an orange-and-mustard
% stripe, and a second colour a hair out of register.
%
% lineup_card.py writes channels.tex (the effective date and every block of
% channels) next to this file and builds it with Tectonic (XeTeX). For the
% print version it also sets \cardstockfalse: white paper, ink only where
% there is type, a stripe or a rule.

% -- the card stock -----------------------------------------------------------
% Darker toward the edges, flecked, and framed by a rounded double rule.
\newif\ifcardstock
\cardstocktrue
\AddToShipoutPictureBG{\AtPageLowerLeft{%
  \begin{tikzpicture}[x=1pt, y=1pt]
    \useasboundingbox (0,0) rectangle (\paperwidth, \paperheight);
    \ifcardstock
    % Solid fills only: xdvipdfmx loses both the opacity and the shadings
    % that pgf puts in a shipout picture. The darkening is nested ellipses.
    \fill[paperedge] (0, 0) rectangle (\paperwidth, \paperheight);
    \foreach \i in {0,...,40} {
      \pgfmathtruncatemacro\edge{100 - 2.5*\i}
      \fill[paperedge!\edge!paper] (0.5\paperwidth, 0.5\paperheight)
        ellipse ({(0.78 - 0.011*\i)*\paperwidth} and {(0.72 - 0.0105*\i)*\paperheight});
    }
    \pgfmathsetseed{1977}
    \foreach \i in {1,...,1400} {
      \fill[paperedge!55!ink]
        ({rnd*\paperwidth}, {rnd*\paperheight}) circle ({0.12 + 0.3*rnd});
    }
    \fi
    \draw[rust, line width=1.6pt, rounded corners=14pt]
      (0.3in, 0.3in) rectangle (\paperwidth - 0.3in, \paperheight - 0.3in);
    \draw[rust, line width=0.5pt, rounded corners=11pt]
      (0.36in, 0.36in) rectangle (\paperwidth - 0.36in, \paperheight - 0.36in);
  \end{tikzpicture}}}

\newcommand\stripes[1][\linewidth]{%
  \begin{tikzpicture}[x=1pt, y=1pt]
    \fill[rust]    (0, 9) rectangle (#1, 13);

\newcommand\misprint[3]{%
  \tikz[baseline=(t.base)]{%
    \node[inner sep=0, text=#2] at (0.9pt, -0.7pt) {#1};
    \node[inner sep=0, text=#3] (t) at (0, 0) {#1};}}

\newcommand\listingsize{\fontsize{8.1}{8.85}\selectfont}

% \block{name}{numbers}: a reversed-out heading bar over its channels. A
% long name wraps beside its numbers.
\newcommand\block[2]{%
  \par\addvspace{5pt}%
  \colorbox{ink}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
    \gothic\bfseries\fontsize{7.2}{8.6}\selectfont\color{paper}%
    \sbox0{#2}%
    \parbox[b]{\dimexpr\linewidth-\wd0-0.5em\relax}{\raggedright
      \addfontfeatures{LetterSpace=5}\strut\MakeUppercase{#1}\strut}%
    \hfill\usebox0}}\par\nobreak\vskip 2pt\nobreak}

% \chan{number}{name}{note}{antenna}: the note sits beside the name, the
% station's own over-the-air channel out at the right margin.
\newcommand\chan[4]{%
  \hangindent=2.25em
  \makebox[1.75em][r]{\gothic\bfseries\color{rust}#1}\hspace{0.5em}#2%
  \if\relax\detokenize{#3}\relax\else\ {\fontsize{6.4}{8}\selectfont\gothic\bfseries #3}\fi
  \if\relax\detokenize{#4}\relax\else
    \leaders\hbox to 0.42em{\hss.\hss}\hskip 0.6em plus 1filll
    \hbox{\typewriter\bfseries\fontsize{7.4}{8}\selectfont #4}\fi
  \par}
